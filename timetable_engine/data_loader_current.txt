# ============================================================
# FIND LATEST SUBMISSION
# ============================================================

def find_latest_submission():

    if not SUBMISSIONS_DIR.exists():

        raise FileNotFoundError(
            f"Submission directory not found:\n"
            f"{SUBMISSIONS_DIR}"
        )

    submissions = [

        folder

        for folder in SUBMISSIONS_DIR.iterdir()

        if (
            folder.is_dir()
            and
            (folder / "submission.json").exists()
        )
    ]

    if not submissions:

        raise FileNotFoundError(
            "No submitted timetable form was found."
        )

    return max(
        submissions,
        key=lambda folder: (
            folder / "submission.json"
        ).stat().st_mtime
    )


# ============================================================
# LOAD SUBMISSION
# ============================================================

def load_submission():

    submission_folder = find_latest_submission()

    submission_file = (
        submission_folder
        / "submission.json"
    )

    with open(
        submission_file,
        "r",
        encoding="utf-8"
    ) as file:

        return json.load(file)


# ============================================================
# FACULTY DATA
# ============================================================

def load_faculty_data():

    submission = load_submission()

    faculty_data = {}

    for faculty in submission.get(
        "faculty",
        []
    ):

        code = str(
            faculty.get(
                "code",
                ""
            )
        ).strip()

        name = str(
            faculty.get(
                "name",
                ""
            )
        ).strip()

        if not code:
            continue

        faculty_data[code] = {

            "name":
                name or code,

            "theory":
                faculty.get(
                    "theory",
                    []
                ) or [],

            "tutorial":
                faculty.get(
                    "tutorial",
                    []
                ) or [],

            "practical":
                faculty.get(
                    "practical",
                    []
                ) or [],
        }

    return faculty_data


# ============================================================
# FACULTY HELPERS
# ============================================================

def get_faculty_for_theory(subject):

    result = []

    for code, data in load_faculty_data().items():

        if subject in data["theory"]:

            result.append(code)

    return result


def get_faculty_for_tutorial(subject):

    result = []

    for code, data in load_faculty_data().items():

        if subject in data["tutorial"]:

            result.append(code)

    return result


def get_faculty_for_practical(subject):

    result = []

    for code, data in load_faculty_data().items():

        if subject in data["practical"]:

            result.append(code)

    return result


def get_faculty_for_subject(subject):

    result = set()

    result.update(
        get_faculty_for_theory(subject)
    )

    result.update(
        get_faculty_for_tutorial(subject)
    )

    result.update(
        get_faculty_for_practical(subject)
    )

    return sorted(result)


def get_subjects_for_faculty(faculty_code):

    faculty = load_faculty_data()

    if faculty_code not in faculty:

        return []

    data = faculty[faculty_code]

    return sorted(
        set(
            data["theory"]
            +
            data["tutorial"]
            +
            data["practical"]
        )
    )

def load_batch_data():

    submission = load_submission()

    batches = {}

    for batch in submission.get(
        "batches",
        []
    ):

        code = str(
            batch.get(
                "code",
                ""
            )
        ).strip()

        if not code:
            continue

        try:

            students = int(
                batch.get(
                    "students",
                    0
                )
            )

        except (
            ValueError,
            TypeError
        ):

            students = 0

        batches[code] = {

            "students":
                students,

            "elective":
                str(
                    batch.get(
                        "elective",
                        ""
                    )
                ).strip(),
        }

    return batches

# ============================================================
# WORKLOAD DATA
#
# Excel is the source of workload information.
# ============================================================

def load_workload():

    if not WORKLOAD_FILE.exists():

        raise FileNotFoundError(
            f"Workload file not found:\n"
            f"{WORKLOAD_FILE}"
        )

    df = pd.read_excel(
        WORKLOAD_FILE
    )

    if df.empty:

        raise ValueError(
            "Workload Excel file is empty."
        )

    # --------------------------------------------------------
    # Normalize column names
    # --------------------------------------------------------

    df.columns = [
        str(column).strip()
        for column in df.columns
    ]

    required_columns = [

        "Course Code",

        "Subject",

        "Theory (hrs)",

        "Tutorial (hrs)",

        "Practical (hrs)",

        "Credits",

        "Session Packaging",
    ]

    missing_columns = [

        column

        for column in required_columns

        if column not in df.columns
    ]

    if missing_columns:

        raise ValueError(
            "Workload file is missing required "
            "columns:\n"
            +
            "\n".join(
                f"- {column}"
                for column in missing_columns
            )
        )

    # --------------------------------------------------------
    # Remove TOTAL row
    # --------------------------------------------------------

    df = df[
        df["Course Code"].notna()
    ].copy()

    df = df[
        df["Course Code"].astype(str).str.upper()
        != "TOTAL"
    ].copy()

    # --------------------------------------------------------
    # Normalize text columns
    # --------------------------------------------------------

    df["Course Code"] = (
        df["Course Code"]
        .astype(str)
        .str.strip()
    )

    df["Subject"] = (
        df["Subject"]
        .astype(str)
        .str.strip()
    )

    df["Session Packaging"] = (
        df["Session Packaging"]
        .fillna("")
        .astype(str)
        .str.strip()
    )

    # --------------------------------------------------------
    # Normalize numeric columns
    # --------------------------------------------------------

    numeric_columns = [

        "Theory (hrs)",

        "Tutorial (hrs)",

        "Practical (hrs)",

        "Credits",
    ]

    for column in numeric_columns:

        df[column] = pd.to_numeric(
            df[column],
            errors="coerce"
        ).fillna(0)

    # --------------------------------------------------------
    # Build clean workload records
    # --------------------------------------------------------

    workload_records = []

    for _, row in df.iterrows():

        workload_records.append({

            "course_code":
                row["Course Code"],

            "subject":
                row["Subject"],

            "theory_hours":
                int(row["Theory (hrs)"]),

            "tutorial_hours":
                int(row["Tutorial (hrs)"]),

            "practical_hours":
                int(row["Practical (hrs)"]),

            "credits":
                float(row["Credits"]),

            "session_packaging":
                row["Session Packaging"],
        })

    # --------------------------------------------------------
    # Validate workload
    # --------------------------------------------------------

    for record in workload_records:

        total_hours = (

            record["theory_hours"]

            +

            record["tutorial_hours"]

            +

            record["practical_hours"]
        )

        if total_hours <= 0:

            raise ValueError(
                f"Subject has zero workload: "
                f"{record['subject']}"
            )

    return {

        "dataframe":
            df,

        "records":
            workload_records,

        "total_theory_hours":
            sum(
                record["theory_hours"]
                for record in workload_records
            ),

        "total_tutorial_hours":
            sum(
                record["tutorial_hours"]
                for record in workload_records
            ),

        "total_practical_hours":
            sum(
                record["practical_hours"]
                for record in workload_records
            ),

        "total_weekly_hours":
            sum(
                record["theory_hours"]
                +
                record["tutorial_hours"]
                +
                record["practical_hours"]

                for record in workload_records
            ),
    }

def build_subject_batches():

    subject_batches = {}

    for subject, definition in SUBJECT_DEFINITIONS.items():

        batches = []

        if "batch_groups" in definition:

            batches.extend(
                definition[
                    "batch_groups"
                ]
            )

        if "practical_groups" in definition:

            batches.extend(
                definition[
                    "practical_groups"
                ]
            )

        if "theory_group" in definition:

            batches.extend(
                TEACHING_GROUPS[
                    definition[
                        "theory_group"
                    ]
                ]
            )

        if "tutorial_group" in definition:

            batches.extend(
                TEACHING_GROUPS[
                    definition[
                        "tutorial_group"
                    ]
                ]
            )

        if "elective_groups" in definition:

            for group_batches in definition[
                "elective_groups"
            ].values():

                batches.extend(
                    group_batches
                )

        subject_batches[subject] = sorted(
            set(batches)
        )

    return subject_batches

def get_workload_record(
    subject,
    workload,
):

    for record in workload["records"]:

        if record["subject"] == subject:

# SESSION CREATION HELPER
# ============================================================

def make_session(
    session_id,
    subject,
    session_type,
    duration,
    group,
    batches,
    elective=None,
):

    session = {

        "id":
            session_id,

        "subject":
            subject,

        "type":
            session_type,

        "duration":
            duration,

        "group":
            group,

        "batch":
            batches,
    }

    if elective:

        session["elective"] = elective

    return session

# COMMON THEORY SESSION CREATOR
# ============================================================

def create_common_theory_sessions(
    subject,
    hours,
    group="COMMON_TE",
):

    sessions = []

    batches = (
        TEACHING_GROUPS[group]
    )

    for index in range(
        1,
        hours + 1
    ):

        sessions.append(
            make_session(

                session_id=(
                    f"{subject}_T{index}"

                subject=subject,

                session_type="theory",

                duration=
                    SCHEDULING_RULES[
                        "theory_duration"
                    ],

                group=group,

                batches=batches.copy(),
            )
        )

    return sessions

# ============================================================
# ELECTIVE THEORY SESSION CREATOR
# ============================================================

def create_elective_theory_sessions(
    subject,
    hours,
    elective_groups,
):

    sessions = []

    for elective, batches in (
        elective_groups.items()
    ):

        if batches == [
            "T1",
            "T2",
        ]:

            group_name = "CLOUD"

        else:

            group_name = "AJP"

        for index in range(
            1,
            hours + 1
        ):

            sessions.append(
                make_session(

                    session_id=(
                        f"{subject}_"
                        f"{elective.replace(' ', '_')}_"
                        f"{index}"
                    ),

                    subject=subject,

                    session_type="theory",

                    duration=
                        SCHEDULING_RULES[
                            "theory_duration"
                        ],

                    group=group_name,

                    batches=batches.copy(),

                    elective=elective,
                )
            )

    return sessions


# ============================================================
# PRACTICAL SESSION CREATOR
# ============================================================

def create_practical_sessions(
    subject,
    blocks,
    groups,
):

    sessions = []

    for block in range(
        1,
        blocks + 1
    ):

        for batch in groups:

            elective = None

            if subject == (
                "Programme Elective Lab"
            ):

                if batch in [
                    "T1",
                    "T2",
                ]:

                    elective = (
                        "Cloud Computing"
                    )

                elif batch == "T3":

                    elective = (
                        "Advanced Java Programming"
                    )

            sessions.append(
                make_session(

                    session_id=(
                        f"{subject}_"
                        f"{block}_"
                        f"{batch}"
                    ),

                    subject=subject,

                    session_type="practical",

                    duration=
                        SCHEDULING_RULES[
                            "practical_duration"
                        ],

                    group=batch,

                    batches=[
                        batch
                    ],

                    elective=elective,
                )
            )

    return sessions


# ============================================================
# MASTER SESSION GENERATOR
# ============================================================

def create_sessions(
    workload,
):

    sessions = []

    # --------------------------------------------------------
    # Process subjects using workload Excel.
    # --------------------------------------------------------

    for subject, definition in (
        SUBJECT_DEFINITIONS.items()
    ):

        workload_record = (
            get_workload_record(
                subject,
                workload
            )
        )

        if workload_record is None:

            raise ValueError(
                f"Subject '{subject}' "
                f"was not found in workload Excel."
            )

        theory_hours = (
            workload_record[
                "theory_hours"
            ]
        )

        tutorial_hours = (
            workload_record[
                "tutorial_hours"
            ]
        )

        practical_hours = (
            workload_record[
                "practical_hours"
            ]
        )

        # ----------------------------------------------------
        # COMMON THEORY
        # ----------------------------------------------------

        if theory_hours > 0:

            if (
                "theory_group"
                in definition
            ):

                sessions.extend(
                    create_common_theory_sessions(
                        subject,
                        theory_hours,
                        definition[
                            "theory_group"
                        ],
                    )
                )

        # ----------------------------------------------------
        # ELECTIVE THEORY
        # ----------------------------------------------------

        if theory_hours > 0:

            if (
                "elective_groups"
                in definition
            ):

                sessions.extend(
                    create_elective_theory_sessions(
                        subject,
                        theory_hours,
                        definition[
                            "elective_groups"
                        ],
                    )
                )

        # ----------------------------------------------------
        # COMMON TUTORIAL
        # ----------------------------------------------------

        if tutorial_hours > 0:

            if (
                "tutorial_group"
                in definition
            ):

                sessions.extend(
                    create_common_tutorial_sessions(
                        subject,
                        tutorial_hours,
                        definition[
                            "tutorial_group"
                        ],
                    )
                )

        # ----------------------------------------------------
        # PRACTICAL
        # ----------------------------------------------------

        if practical_hours > 0:

            # Programme Elective Lab uses elective groups
            # rather than the generic practical_groups key.
            # Each batch gets one 2-hour lab session.
            if (
                subject == "Programme Elective Lab"
                and "elective_groups" in definition
            ):

                for elective, groups in definition[
                    "elective_groups"
                ].items():

                    for batch in groups:

                        sessions.append(
                            make_session(
                                session_id=(
                                    f"{subject}_"
                                    f"1_{batch}"
                                ),
                                subject=subject,
                                session_type="practical",
                                duration=SCHEDULING_RULES[
                                    "practical_duration"
                                ],
                                group=batch,
                                batches=[batch],
                                elective=elective,
                            )
                        )

            elif "practical_groups" in definition:

                practical_blocks = int(
                    definition.get(
                        "practical_blocks",
                        practical_hours // 2
                    )
                )

                sessions.extend(
                    create_practical_sessions(
                        subject,
                        practical_blocks,
                        definition[
                            "practical_groups"
                        ],
                    )
                )

    return sessions


# ============================================================
# VALIDATE GENERATED SESSIONS
# ============================================================

def validate_sessions(
    sessions,
    batches,
    workload,
):

    errors = []

    # --------------------------------------------------------
    # SESSION COUNT
    # --------------------------------------------------------

    if len(sessions) != 37:

        errors.append(
            f"Expected 37 sessions, "
            f"but generated {len(sessions)}."
        )

    # --------------------------------------------------------
    # WORKLOAD TOTAL
    # --------------------------------------------------------

    if workload[
        "total_weekly_hours"
    ] != 28:

        errors.append(
            "Expected 28 weekly workload hours, "
            f"but found "
            f"{workload['total_weekly_hours']}."
        )

    # --------------------------------------------------------
    # REQUIRED BATCHES
    # --------------------------------------------------------

    required_batches = {
        "T1",
        "T2",
        "T3",
    }

    available_batches = set(
        batches.keys()
    )

    missing_batches = (
        required_batches
        -
        available_batches
    )

    if missing_batches:

        errors.append(
            "Missing required batches: "
            +
            ", ".join(
                sorted(
                    missing_batches
                )
            )
        )

    # --------------------------------------------------------
    # SESSION STRUCTURE
    # --------------------------------------------------------

    for session in sessions:

        required_keys = [

            "id",

            "subject",

            "type",

            "duration",

            "group",

            "batch",
        ]

        for key in required_keys:

            if key not in session:

                errors.append(
                    f"Session "
                    f"{session.get('id', 'UNKNOWN')} "
                    f"is missing '{key}'."
                )

        if session.get(
            "duration"
        ) not in [
            1,
            2,
        ]:

            errors.append(
                f"Invalid duration in "
                f"{session.get('id', 'UNKNOWN')}: "
                f"{session.get('duration')}"
            )

        if session.get(
            "type"
        ) == "practical":

            if session.get(
                "duration"
            ) != 2:

                errors.append(
                    f"Practical session "
                    f"{session.get('id')} "
                    f"must be 2 hours."
                )

        for batch in session.get(
            "batch",
            []
        ):

            if batch not in batches:

                errors.append(
                    f"Session "
                    f"{session.get('id')} "
                    f"references unknown "
                    f"batch {batch}."
                )

    return errors
